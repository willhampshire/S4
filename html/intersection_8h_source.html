46// Computes the intersection of (assumed) convex polygons P and Q.
47// |P| == n, |Q| == m, returns Pi = intersection(P,Q), |Pi| == *ni
48// On entry, *ni should be the size of Pi.
49// If Pi is too small, returns -10, and *ni is the needed size.
50// If P is in Q, returns 1.
51// If Q is in P, returns 2.
52// Otherwise, on success, returns 0.
53// Returns -n if there is an error with the n-th argument.
54int convex_polygon_intersection(
55 int n, // n >= 3
56 const double *P,
57 int m, // m >= 3
58 const double *Q,
59 int *ni, // on input, size of Pi (length/2), on output, numer of points in Pi
60 double *Pi // output intersection polygon
61);
