61template <class T>
62inline T Pythag2(T x, T y){ // dlapy2
63 using namespace std;
64 // returns sqrt(x**2+y**2), taking care not to cause unnecessary overflow.
65 T a = abs(x);
66 T b = abs(y);
67 if(a < b){ swap(a,b); }
68 if(0 == b){
69 return a;
70 }else{
71 b /= a;
72 return a * sqrt(b*b + T(1));
73 }
74}

76template <class T>
77inline T Pythag3(T x, T y, T z){ // dlapy3
78 using namespace std;
79 // returns sqrt(x**2+y**2+z**2), taking care not to cause unnecessary overflow.
80 T xabs = abs(x);
81 T yabs = abs(y);
82 T zabs = abs(z);
83 T w = xabs;
84 if(yabs > w){ w = yabs; }
85 if(zabs > w){ w = zabs; }
86 if(0 == w){
87 // W can be zero for max(0,nan,0)
88 // adding all three entries together will make sure
89 // NaN will not disappear.
90 return xabs + yabs + zabs;
91 }else{
92 xabs /= w;
93 yabs /= w;
94 zabs /= w;
95 return w * sqrt(xabs*xabs + yabs*yabs + zabs*zabs);
96 }
97}

99template <class T>
100void ComplexDivide(const std::complex<T> &num, const std::complex<T> &den, std::complex<T> *result){ // zladiv, dladiv, cladiv, sladiv
101 // The algorithm is due to Robert L. Smith and can be found
102 // in D. Knuth, The art of Computer Programming, Vol.2, p.195
103 using namespace std;
104 if(abs(den.imag()) < abs(den.real())){
105 T e = den.imag()/den.real();
106 T f = den.real() + den.imag()*e;
107 *result = std::complex<double>(
108 (num.real() + num.imag()*e) / f,
109 (num.imag() - num.real()*e) / f
110 );
111 }else{
112 T e = den.real()/den.imag();
113 T f = den.imag() + den.real()*e;
114 *result = std::complex<double>(
115 (num.imag() + num.real()*e) / f,
116 (num.imag()*e - num.real()) / f
117 );
118 }

168template <char norm='M'>
169struct CheapHessenbergNorm{ // zlanhs, dlanhs, clanhs, slanhs
170 template <class T>
171 CheapHessenbergNorm(size_t n, const T *a, size_t lda, typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type *result, typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type *work = NULL){
172 typedef typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type real_type;
173 *result = 0;
174 if(n < 1){ return; }
175 if('M' == norm){ // max(abs(A(i,j)))
176 for(size_t j = 0; j < n; ++j){
177 size_t ilimit = j+2; if(n < ilimit){ ilimit = n; }
178 for(size_t i = 0; i < ilimit; ++i){
179 real_type ca = RNP::TBLAS::_RealOrComplexChooser<T>::_abs(a[i+j*lda]);
180 if(ca > *result){ *result = ca; }
181 }
182 }
183 }else if('O' == norm || '1' == norm){ // max col sum
184 for(size_t j = 0; j < n; ++j){
185 size_t ilimit = j+2; if(n < ilimit){ ilimit = n; }
186 real_type sum = 0;
187 for(size_t i = 0; i < ilimit; ++i){
188 sum += RNP::TBLAS::_RealOrComplexChooser<T>::_abs(a[i+j*lda]);
189 }
190 if(sum > *result){ *result = sum; }
191 }
192 }else if('I' == norm){ // max row sum
193 if(NULL == work){ // can't accumulate row sums
194 for(size_t i = 0; i < n; ++i){
195 size_t jstart = 0; if(i > 0){ jstart = i-1; }
196 real_type sum = 0;
197 for(size_t j = 0; j < n; ++j){
198 sum += RNP::TBLAS::_RealOrComplexChooser<T>::_abs(a[i+j*lda]);
199 }
200 if(sum > *result){ *result = sum; }
201 }
202 }else{ // accumulate row sums in a cache-friendlier traversal order
203 for(size_t i = 0; i < n; ++i){ work[i] = 0; }
204 for(size_t j = 0; j < n; ++j){
205 size_t ilimit = j+2; if(n < ilimit){ ilimit = n; }
206 for(size_t i = 0; i < ilimit; ++i){
207 work[i] += RNP::TBLAS::_RealOrComplexChooser<T>::_abs(a[i+j*lda]);
208 }
209 }
210 for(size_t i = 0; i < n; ++i){
211 if(work[i] > *result){ *result = work[i]; }
212 }
213 }
214 }else if('F' == norm || 'E' == norm){ // Frobenius norm
215 using namespace std;
216#ifdef RNP_NUMERIC_ROBUST
217 real_type scale = 0;
218 real_type sum = 1;
219 for(size_t j = 0; j < n; ++j){
220 size_t ilimit = j+2; if(n < ilimit){ ilimit = n; }
221 real_type sum = 0;
222 for(size_t i = 0; i < ilimit; ++i){
223 real_type ca = RNP::TBLAS::_RealOrComplexChooser<T>::_abs(a[i+j*lda]);
224 if(scale < ca){
225 real_type r = scale/ca;
226 sum = real_type(1) + sum*r*r;
227 scale = ca;
228 }else{
229 real_type r = ca/scale;
230 sum += r*r;
231 }
232 }
233 }
234 *result = scale*sqrt(sum);
235#else
236 real_type sum = 0;
237 for(size_t j = 0; j < n; ++j){
238 size_t ilimit = j+2; if(n < ilimit){ ilimit = n; }
239 real_type sum = 0;
240 for(size_t i = 0; i < ilimit; ++i){
241 sum += RNP::TBLAS::_RealOrComplexChooser<T>::_abs2(a[i+j*lda]);
242 }
243 }
244 *result = sqrt(sum);
245#endif
246 }
247 }

327template <char type='G'> // zlascl, dlascl, clascl, slascl
328struct RescaleMatrix{
329 template <class TS, class T>
330 RescaleMatrix(size_t kl, size_t ku, const TS &cfrom, const TS &cto, size_t m, size_t n, T *a, size_t lda){
331 // Purpose
332 // =======
333
334 // ZLASCL multiplies the M by N complex matrix A by the real scalar
335 // CTO/CFROM. This is done without over/underflow as long as the final
336 // result CTO*A(I,J)/CFROM does not over/underflow. TYPE specifies that
337 // A may be full, upper triangular, lower triangular, upper Hessenberg,
338 // or banded.
339
340 // Arguments
341 // =========
342
343 // TYPE (input) CHARACTER*1
344 // TYPE indices the storage type of the input matrix.
345 // = 'G': A is a full matrix.
346 // = 'L': A is a lower triangular matrix.
347 // = 'U': A is an upper triangular matrix.
348 // = 'H': A is an upper Hessenberg matrix.
349 // = 'B': A is a symmetric band matrix with lower bandwidth KL
350 // and upper bandwidth KU and with the only the lower
351 // half stored.
352 // = 'Q': A is a symmetric band matrix with lower bandwidth KL
353 // and upper bandwidth KU and with the only the upper
354 // half stored.
355 // = 'Z': A is a band matrix with lower bandwidth KL and upper
356 // bandwidth KU.
357
358 // KL (input) int
359 // The lower bandwidth of A. Referenced only if TYPE = 'B',
360 // 'Q' or 'Z'.
361
362 // KU (input) int
363 // The upper bandwidth of A. Referenced only if TYPE = 'B',
364 // 'Q' or 'Z'.
365
366 // CFROM (input) DOUBLE PRECISION
367 // CTO (input) DOUBLE PRECISION
368 // The matrix A is multiplied by CTO/CFROM. A(I,J) is computed
369 // without over/underflow if the final result CTO*A(I,J)/CFROM
370 // can be represented without over/underflow. CFROM must be
371 // nonzero.
372
373 // M (input) int
374 // The number of rows of the matrix A. M >= 0.
375
376 // N (input) int
377 // The number of columns of the matrix A. N >= 0.
378
379 // A (input/output) COMPLEX*16 array, dimension (LDA,N)
380 // The matrix to be multiplied by CTO/CFROM. See TYPE for the
381 // storage type.
382
383 // LDA (input) int
384 // The leading dimension of the array A. LDA >= max(1,M).
385
386 if(n == 0 || m == 0){ return; }
387
388 const TS smlnum = std::numeric_limits<TS>::min();
389 const TS bignum = 1. / smlnum;
390
391 TS cfromc = cfrom;
392 TS ctoc = cto;
393
394 bool done = true;
395 do{
396 const TS cfrom1 = cfromc * smlnum;
397 TS mul;
398 if(cfrom1 == cfromc){
399 // CFROMC is an inf. Multiply by a correctly signed zero for
400 // finite CTOC, or a NaN if CTOC is infinite.
401 mul = ctoc / cfromc;
402 done = true;
403 //cto1 = ctoc;
404 }else{
405 const TS cto1 = ctoc / bignum;
406 if(cto1 == ctoc){
407 // CTOC is either 0 or an inf. In both cases, CTOC itself
408 // serves as the correct multiplication factor.
409 mul = ctoc;
410 done = true;
411 //cfromc = 1.;
412 }else if(std::abs(cfrom1) > std::abs(ctoc) && ctoc != 0.){
413 mul = smlnum;
414 done = false;
415 cfromc = cfrom1;
416 }else if(std::abs(cto1) > std::abs(cfromc)){
417 mul = bignum;
418 done = false;
419 ctoc = cto1;
420 }else{
421 mul = ctoc / cfromc;
422 done = true;
423 }
424 }
425
426 if(type == 'G'){ // Full matrix
427 for(size_t j = 0; j < n; ++j){
428 for(size_t i = 0; i < m; ++i){
429 a[i+j*lda] *= mul;
430 }
431 }
432 }else if(type == 'L'){ // Lower triangular matrix
433 for(size_t j = 0; j < n; ++j){
434 for(size_t i = j; i < m; ++i){
435 a[i+j*lda] *= mul;
436 }
437 }
438 }else if(type == 'U'){
439 // Upper triangular matrix
440 for(size_t j = 0; j < n; ++j){
441 size_t ilimit = j+1; if(m < ilimit){ ilimit = m; }
442 for(size_t i = 0; i < ilimit; ++i){
443 a[i+j*lda] *= mul;
444 }
445 }
446 }else if(type == 'H'){
447 // Upper Hessenberg matrix
448 for(size_t j = 0; j < n; ++j){
449 size_t ilimit = j+2; if(m < ilimit){ ilimit = m; };
450 for(size_t i = 0; i < ilimit; ++i){
451 a[i+j*lda] *= mul;
452 }
453 }
454 }else if(type == 'B'){
455 // Lower half of a symmetric band matrix
456 for(size_t j = 0; j < n; ++j){
457 size_t ilimit = n-j; if(kl+1 < ilimit){ ilimit = kl+1; }
458 for(size_t i = 0; i < ilimit; ++i){
459 a[i+j*lda] *= mul;
460 }
461 }
462 }else if(type == 'Q'){
463 // Upper half of a symmetric band matrix
464 for(size_t j = 0; j < n; ++j){
465 size_t istart = (ku > j) ? ku-j : 0;
466 for(size_t i = istart; i <= ku; ++i){
467 a[i+j*lda] *= mul;
468 }
469 }
470 }else if(type == 'Z'){
471 // Band matrix
472 size_t k3 = 2*kl + ku + 1;
473 for(size_t j = 0; j < n; ++j){
474 size_t istart = kl+ku-j;
475 if(kl > istart){ istart = kl; }
476 size_t ilimit = kl + ku + m-j;
477 if(k3 < ilimit){ ilimit = k3; }
478 for(size_t i = istart; i < ilimit; ++i){
479 a[i+j*lda] *= mul;
480 }
481 }
482 }
483 }while(!done);
484 }
485};

487template <char side='L'>
488struct ApplyElementaryReflector{ // zlarf, dlarf, clarf, slarf
489 template <class T>
490 ApplyElementaryReflector(size_t m, size_t n, const T *v, size_t incv, const T &tau, T *c, size_t ldc, T *work){
491 // Applies an elementary reflector H to an M-by-N matrix C,
492 // from either the left or the right. H is represented in the form
493 // H = I - tau * v * v'
494 // where tau is a complex scalar and v is a complex vector.
495 //
496 // To apply H' (the conjugate transpose of H), supply conj(tau) instead
497
498 // Arguments
499 // =========
500 //
501 // SIDE = 'L': form H * C
502 // = 'R': form C * H
503 //
504 // M The number of rows of the matrix C.
505 //
506 // N The number of columns of the matrix C.
507 //
508 // V (1 + (M-1)*abs(INCV)) if SIDE = 'L'
509 // or (1 + (N-1)*abs(INCV)) if SIDE = 'R'
510 // The vector v in the representation of H. V is not used if
511 // TAU = 0.
512 //
513 // INCV The increment between elements of v. INCV <> 0.
514 //
515 // TAU The value tau in the representation of H.
516 //
517 // C (input/output) COMPLEX*16 array, dimension (LDC,N)
518 // On entry, the M-by-N matrix C.
519 // On exit, C is overwritten by the matrix H * C if SIDE = 'L',
520 // or C * H if SIDE = 'R'.
521 //
522 // LDC The leading dimension of the array C. LDC >= max(1,M).
523 //
524 // WORK (workspace) COMPLEX*16 array, dimension
525 // (N) if SIDE = 'L'
526 // or (M) if SIDE = 'R'
527
528 size_t lenv = 0;
529 size_t lenc = 0;
530 if(T(0) != tau){
531 // Set up variables for scanning V. LASTV begins pointing to the end of V.
532 if('L' == side){
533 lenv = m;
534 }else{
535 lenv = n;
536 }
537 size_t i = 0;
538 if(incv > 0){
539 i = (lenv - 1) * incv;
540 }
541 // Look for the last non-zero row in V.
542 while(lenv > 0 && (T(0) == v[i])){
543 --lenv;
544 i -= incv;
545 }
546 if('L' == side){ // Scan for the last non-zero column in C(1:lastv,:).
547 lenc = 1+RNP::TLASupport::LastNonzeroColumn(lenv, n, c, ldc);
548 }else{ // Scan for the last non-zero row in C(:,1:lastv).
549 lenc = 1+RNP::TLASupport::LastNonzeroRow(m, lenv, c, ldc);
550 }
551 }
552
553 if('L' == side){ // Form H * C
554 if(lenv > 0 && lenc > 0){
555 // w(1:lastc,1) := C(1:lastv,1:lastc)' * v(1:lastv,1)
556 RNP::TBLAS::MultMV<'C'>(lenv, lenc, T(1), c, ldc, v, incv, T(0), work, 1);
557 // C(1:lastv,1:lastc) := C(...) - v(1:lastv,1) * w(1:lastc,1)'
558 RNP::TBLAS::ConjugateRank1Update(lenv, lenc, -tau, v, incv, work, 1, c, ldc);
559 }
560 }else{ // Form C * H
561 if(lenv > 0 && lenc > 0){
562 // w(1:lastc,1) := C(1:lastc,1:lastv) * v(1:lastv,1)
563 RNP::TBLAS::MultMV<'N'>(lenc, lenv, T(1), c, ldc, v, incv, T(0), work, 1);
564 // C(1:lastc,1:lastv) := C(...) - w(1:lastc,1) * v(1:lastv,1)'
565 RNP::TBLAS::ConjugateRank1Update(lenc, lenv, -tau, work, 1, v, incv, c, ldc);
566 }
567 }
568 }
569};

571template <char side, char trans, char direct, char storev>
572struct ApplyElementaryReflectorBlocked{ // zlarfb, dlarfb, clarfb, slarfb
573 template <class T>
574 ApplyElementaryReflectorBlocked(size_t m, size_t n, size_t k, const T *v, size_t ldv, const T *t, size_t ldt, T *c, size_t ldc, T *work, size_t ldwork){
575 // ZLARFB applies a complex block reflector H or its transpose H' to a
576 // complex M-by-N matrix C, from either the left or the right.
577 //
578 // Arguments
579 // =========
580 //
581 // SIDE (input) CHARACTER*1
582 // = 'L': apply H or H' from the Left
583 // = 'R': apply H or H' from the Right
584 //
585 // TRANS (input) CHARACTER*1
586 // = 'N': apply H (No transpose)
587 // = 'C': apply H' (Conjugate transpose)
588 //
589 // DIRECT (input) CHARACTER*1
590 // Indicates how H is formed from a product of elementary
591 // reflectors
592 // = 'F': H = H(1) H(2) . . . H(k) (Forward)
593 // = 'B': H = H(k) . . . H(2) H(1) (Backward)
594 //
595 // STOREV (input) CHARACTER*1
596 // Indicates how the vectors which define the elementary
597 // reflectors are stored:
598 // = 'C': Columnwise
599 // = 'R': Rowwise

1020 typedef typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type real_type;
1021
1022 // Purpose
1023 // =======
1024 //
1025 // ZLARFG generates a complex elementary reflector H of order n, such
1026 // that
1027 //
1028 // H' * ( alpha ) = ( beta ), H' * H = I.
1029 // ( x ) ( 0 )
1030 //
1031 // where alpha and beta are scalars, with beta real, and x is an
1032 // (n-1)-element complex vector. H is represented in the form
1033 //
1034 // H = I - tau * ( 1 ) * ( 1 v' ) ,
1035 // ( v )
1036 //
1037 // where tau is a complex scalar and v is a complex (n-1)-element
1038 // vector. Note that H is not hermitian.
1039 //
1040 // If the elements of x are all zero and alpha is real, then tau = 0
1041 // and H is taken to be the unit matrix.
1042 //
1043 // Otherwise 1 <= real(tau) <= 2 and std::abs(tau-1) <= 1 .
1044 //
1045 // Arguments
1046 // =========
1047 //
1048 // N The order of the elementary reflector.
1049 //
1050 // ALPHA (input/output) std::complex<double>
1051 // On entry, the value alpha.
1052 // On exit, it is overwritten with the value beta.
1053 //
1054 // X (input/output) std::complex<double> array, dimension
1055 // (1+(N-2)*std::abs(INCX))
1056 // On entry, the vector x.
1057 // On exit, it is overwritten with the vector v.
1058 //
1059 // INCX The increment between elements of X. INCX > 0.
1060 //
1061 // TAU (output) std::complex<double>
1062 // The value tau.
1063 //
1064 using namespace std;
1065
1066 if(n == 0){
1067 *tau = 0;
1068 return;
1069 }
1070
1071 real_type xnorm = RNP::TBLAS::Norm2( n-1, x, incx );
1072 if(xnorm == 0 && RNP::TBLAS::_RealOrComplexChooser<T>::_imag(*alpha) == 0){ // H = I
1073 *tau = 0;
1074 }else{ // general case
1075 real_type beta = RNP::TLASupport::Pythag3( RNP::TBLAS::_RealOrComplexChooser<T>::_real(*alpha), RNP::TBLAS::_RealOrComplexChooser<T>::_imag(*alpha), xnorm );
1076 if(RNP::TBLAS::_RealOrComplexChooser<T>::_real(*alpha) > 0){ beta = -beta; };
1077 const real_type safmin = std::numeric_limits<real_type>::min() / (2*std::numeric_limits<real_type>::epsilon());
1078 const real_type rsafmn = real_type(1) / safmin;
1079 //

1083 ++knt;

1086 *alpha *= rsafmn;
1087 }while(abs(beta) < safmin );

1096 // If ALPHA is subnormal, it may lose relative accuracy
1097 while(knt --> 0){
1098 beta *= safmin;
1099 }
1100 *alpha = beta;

1111 // =======

1128 // =========

1134 // On exit, it is overwritten with the value beta.

1147 return;

1167 x += incx;

1170 }

1178 x += incx;

1200 beta = RNP::TLASupport::Pythag2(amag, xnorm); if(RNP::TBLAS::_RealOrComplexChooser<T>::_real(*alpha) < 0){ beta = -beta; }
1201 }
1202 *alpha += beta;
1203 if(beta < 0){
1204 beta = -beta;
1205 }else{
1206 // The following original 3 lines is specific to complex numbers
1207 //alphr = alphi * (alphi / alpha->real()) + xnorm * (xnorm / alpha->real());
1208 //*tau = std::complex<double>(alphr / beta, -alphi / beta);
1209 //*alpha = std::complex<double>(-alphr, alphi);

1356void GenerateOrthognalMatrixFromElementaryReflectors(size_t m, size_t n, size_t k,
1357 T *a, int lda, const T *tau, T *work)
1358{
1359 using namespace std;
1360
1361 // ZUNG2R generates an m by n complex matrix Q with orthonormal columns,
1362 // which is defined as the first n columns of a product of k elementary
1363 // reflectors of order m
1364 // Q = H(1) H(2) . . . H(k)
1365 // as returned by ZGEQRF.
1366
1367 // Arguments
1368 // =========
1369
1370 // M The number of rows of the matrix Q. M >= 0.
1371
1372 // N The number of columns of the matrix Q. M >= N >= 0.
1373
1374 // K The number of elementary reflectors whose product defines the
1375 // matrix Q. N >= K >= 0.
1376
1377 // A (input/output) COMPLEX*16 array, dimension (LDA,N)
1378 // On entry, the i-th column must contain the vector which
1379 // defines the elementary reflector H(i), for i = 1,2,...,k, as
1380 // returned by ZGEQRF in the first k columns of its array
1381 // argument A.
1382 // On exit, the m by n matrix Q.
1383
1384 // LDA (input) int
1385 // The first dimension of the array A. LDA >= max(1,M).
1386
1387 // TAU (input) COMPLEX*16 array, dimension (K)
1388 // TAU(i) must contain the scalar factor of the elementary
1389 // reflector H(i), as returned by ZGEQRF.
1390
1391 // WORK (workspace) COMPLEX*16 array, dimension (N)
1392
1393 if(n < 1){ return; }
1394
1395 // Initialise columns k+1:n to columns of the unit matrix
1396 for(size_t j = k; j < n; ++j){
1397 for(size_t l = 0; l < m; ++l){
1398 a[l+j*lda] = 0;
1399 }
1400 a[j+j*lda] = 1;
1401 }

1406 a[i+i*lda] = 1;

1408 }
1409 if((size_t)i < m-1){
1410 RNP::TBLAS::Scale(m-i-1, -tau[i], &a[i+1+i*lda], 1);
1411 }
1412 a[i+i*lda] = T(1) - tau[i];
1413
1414 // Set A(1:i-1,i) to zero
1415 for(int l = 0; l < i; ++l){
1416 a[l+i*lda] = 0;
1417 }

1421template <class T> // zgeqrf, zgeqr2, dgeqrf, dgeqr2, cgeqrf, cgeqr2, sgeqrf, sgeqr2
1422void QRFactorization(size_t m, size_t n, T *a, size_t lda, T *tau, T *work){
1423 using namespace std;
1424
1425 // ZGEQR2 computes a QR factorization of a complex m by n matrix A = Q * R.
1426
1427 // Arguments
1428 // =========
1429
1430 // M The number of rows of the matrix A. M >= 0.
1431
1432 // N The number of columns of the matrix A. N >= 0.
1433
1434 // A (input/output) COMPLEX*16 array, dimension (LDA,N)
1435 // On entry, the m by n matrix A.
1436 // On exit, the elements on and above the diagonal of the array
1437 // contain the min(m,n) by n upper trapezoidal matrix R (R is
1438 // upper triangular if m >= n); the elements below the diagonal,
1439 // with the array TAU, represent the unitary matrix Q as a
1440 // product of elementary reflectors (see Further Details).
1441
1442 // LDA The leading dimension of the array A. LDA >= max(1,M).
1443
1444 // TAU (output) COMPLEX*16 array, dimension (min(M,N))
1445 // The scalar factors of the elementary reflectors (see Further
1446 // Details).
1447
1448 // WORK (workspace) COMPLEX*16 array, dimension (N)
1449
1450 // Further Details
1451 // ===============
1452
1453 // The matrix Q is represented as a product of elementary reflectors
1454 // Q = H(1) H(2) . . . H(k), where k = min(m,n).
1455 // Each H(i) has the form
1456 // H(i) = I - tau * v * v'
1457 // where tau is a complex scalar, and v is a complex vector with
1458 // v(1:i-1) = 0 and v(i) = 1; v(i+1:m) is stored on exit in A(i+1:m,i),
1459 // and tau in TAU(i).
1460
1461 size_t k = m; if(n < k){ k = n; }
1462 for(size_t i = 0; i < k; ++i){
1463 // Generate elementary reflector H(i) to annihilate A(i+1:m,i)
1464 size_t row = m-1; if(i+1 < row){ row = i+1; }
1465 RNP::TLASupport::GenerateElementaryReflector(m-i, &a[i+i*lda], &a[row+i*lda], 1, &tau[i]);
1466 if(i < n-1){
1467 // Apply H(i)' to A(i:m,i+1:n) from the left
1468 std::complex<double> alpha = a[i+i*lda];
1469 a[i+i*lda] = 1;
1470 RNP::TLASupport::ApplyElementaryReflector<'L'>(m-i, n-i-1, &a[i+i*lda], 1, std::conj(tau[i]), &a[i+(i+1)*lda], lda, work);
1471 a[i+i*lda] = alpha;
1472 }
1473 }
1474}

1476template <class T> // zlartg, dlartg, clartg, slartg
1477void GeneratePlaneRotation(const T &f, const T &g, typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type *cs, T *sn, T *r)
1478{
1479 typedef typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type real_type;
1480 using namespace std;
1481
1482 // Purpose
1483 // =======
1484
1485 // ZLARTG generates a plane rotation so that
1486
1487 // [ CS SN ] [ F ] [ R ]
1488 // [ __ ] . [ ] = [ ] where CS**2 + |SN|**2 = 1.
1489 // [ -SN CS ] [ G ] [ 0 ]
1490
1491 // This is a faster version of the BLAS1 routine ZROTG, except for
1492 // the following differences:
1493 // F and G are unchanged on return.
1494 // If G=0, then CS=1 and SN=0.
1495 // If F=0, then CS=0 and SN is chosen so that R is real.
1496
1497 // Arguments
1498 // =========
1499
1500 // F (input) COMPLEX*16
1501 // The first component of vector to be rotated.
1502
1503 // G (input) COMPLEX*16
1504 // The second component of vector to be rotated.
1505
1506 // CS (output) DOUBLE PRECISION
1507 // The cosine of the rotation.
1508
1509 // SN (output) COMPLEX*16
1510 // The sine of the rotation.
1511
1512 // R (output) COMPLEX*16
1513 // The nonzero component of the rotated vector.
1514
1515 // Further Details
1516 // ======= =======
1517
1518 // 3-5-96 - Modified with a new algorithm by W. Kahan and J. Demmel
1519
1520 const real_type safmin = std::numeric_limits<real_type>::min();
1521 const real_type eps = std::numeric_limits<real_type>::epsilon();
1522 const real_type safmn2 = sqrt(safmin/eps);
1523 const real_type safmx2 = real_type(1) / safmn2;
1524
1525 real_type aif = RNP::TBLAS::_RealOrComplexChooser<T>::_absinf(f);
1526 real_type aig = RNP::TBLAS::_RealOrComplexChooser<T>::_absinf(g);
1527 real_type scale = ((aif > aig) ? aif : aig);
1528 T fs = f;
1529 T gs = g;
1530 int count = 0;
1531 // Edit by vkl: check for inf; otherwise never terminates
1532 if(scale >= safmx2){
1533 do{
1534 ++count;
1535 fs *= safmn2;
1536 gs *= safmn2;
1537 scale *= safmn2;
1538 }while(scale >= safmx2 && scale <= std::numeric_limits<real_type>::max());
1539 }else if(scale <= safmn2){
1540 if(T(0) == g){
1541 *cs = real_type(1);
1542 *sn = 0;
1543 *r = f;
1544 return;
1545 }
1546 do{
1547 --count;
1548 fs *= safmx2;
1549 gs *= safmx2;
1550 scale *= safmx2;
1551 }while(scale <= safmn2 && scale <= std::numeric_limits<real_type>::max());
1552 }
1553 real_type f2 = RNP::TBLAS::_RealOrComplexChooser<T>::_abs2(fs);
1554 real_type g2 = RNP::TBLAS::_RealOrComplexChooser<T>::_abs2(gs);
1555 real_type f2limit = g2; if(1 > f2limit){ f2limit = 1; }
1556 if(f2 <= f2limit * safmin){
1557 // This is a rare case: F is very small. */
1558 if(T(0) == f){
1559 *cs = 0;
1560 *r = RNP::TBLAS::_RealOrComplexChooser<T>::_abs(g);
1561 // Do complex/real division explicitly with two real divisions */
1562 *sn = gs/abs(gs);
1563 return;
1564 }
1565 real_type f2s = abs(fs);
1566 // G2 and G2S are accurate */
1567 // G2 is at least SAFMIN, and G2S is at least SAFMN2 */
1568 real_type g2s = sqrt(g2);
1569 // Error in CS from underflow in F2S is at most */
1570 // UNFL / SAFMN2 .lt. sqrt(UNFL*EPS) .lt. EPS */
1571 // If MAX(G2,ONE)=G2, then F2 .lt. G2*SAFMIN, */
1572 // and so CS .lt. sqrt(SAFMIN) */
1573 // If MAX(G2,ONE)=ONE, then F2 .lt. SAFMIN */
1574 // and so CS .lt. sqrt(SAFMIN)/SAFMN2 = sqrt(EPS) */
1575 // Therefore, CS = F2S/G2S / sqrt( 1 + (F2S/G2S)**2 ) = F2S/G2S */
1576 *cs = f2s / g2s;
1577 // Make sure abs(FF) = 1 */
1578 // Do complex/real division explicitly with 2 real divisions */
1579 T ff(f);
1580 if(RNP::TBLAS::_RealOrComplexChooser<T>::_absinf(f) <= real_type(1)){
1581 ff *= safmx2;
1582 }
1583 ff /= RNP::TBLAS::_RealOrComplexChooser<T>::_abs(ff);
1584 *sn = ff*RNP::TBLAS::_RealOrComplexChooser<T>::_conj(gs/g2s);
1585 *r = *cs*f + *sn*g;
1586 }else{
1587 // This is the most common case. */
1588 // Neither F2 nor F2/G2 are less than SAFMIN */
1589 // F2S cannot overflow, and it is accurate */
1590 real_type f2s = sqrt(g2 / f2 + real_type(1));
1591 // Do the F2S(real)*FS(complex) multiply with two real multiplies */
1592 *r = f2s*fs;
1593 *cs = real_type(1) / f2s;
1594 // Do complex/real division explicitly with two real divisions */
1595 *sn = (*r/(f2 + g2)) * RNP::TBLAS::_RealOrComplexChooser<T>::_conj(gs);
1596 while(count > 0){
1597 *r *= safmx2;
1598 --count;
1599 }
1600 while(count < 0){
1601 *r *= safmn2;
1602 ++count;
1603 }
1604 }
1605}

1635 // C and S define a rotation
1636 // [ C S ]
1637 // [ -conjg(S) C ]
1638 // where C*C + S*CONJG(S) = 1.0.
1639
1640 for(size_t i = 0; i < n; ++i){
1641 T stemp = c*(*cx) + s*(*cy);
1642 *cy = c*(*cy) - std::conj(s)*(*cx);
1643 *cx = stemp;
1644 cx += incx;
1645 cy += incy;
1646 }

1649template <class T> // zgehd2, dgehd2, cgehd2, sgehd2
1650void HessenbergReductionUnblocked(size_t n, size_t ilo, size_t ihi, T *a, size_t lda, T *tau, T *work){
1651 // ZGEHD2 reduces a complex general matrix A to upper Hessenberg form H
1652 // by a unitary similarity transformation: Q' * A * Q = H .
1653
1654 // Arguments
1655 // =========
1656
1657 // N The order of the matrix A. N >= 0.
1658
1659 // ILO (input) INTEGER
1660 // IHI (input) INTEGER
1661 // It is assumed that A is already upper triangular in rows
1662 // and columns 1:ILO-1 and IHI+1:N. ILO and IHI are normally
1663 // set by a previous call to ZGEBAL; otherwise they should be
1664 // set to 1 and N respectively. See Further Details.
1665 // 1 <= ILO <= IHI <= max(1,N).
1666
1667 // A (input/output) COMPLEX*16 array, dimension (LDA,N)
1668 // On entry, the n by n general matrix to be reduced.
1669 // On exit, the upper triangle and the first subdiagonal of A
1670 // are overwritten with the upper Hessenberg matrix H, and the
1671 // elements below the first subdiagonal, with the array TAU,
1672 // represent the unitary matrix Q as a product of elementary
1673 // reflectors. See Further Details.
1674
1675 // LDA The leading dimension of the array A. LDA >= max(1,N).
1676
1677 // TAU (output) COMPLEX*16 array, dimension (N-1)
1678 // The scalar factors of the elementary reflectors (see Further
1679 // Details).
1680
1681 // WORK (workspace) COMPLEX*16 array, dimension (N)
1682
1683 // Further Details
1684 // ===============
1685
1686 // The matrix Q is represented as a product of (ihi-ilo) elementary
1687 // reflectors
1688
1689 // Q = H(ilo) H(ilo+1) . . . H(ihi-1).
1690
1691 // Each H(i) has the form
1692
1693 // H(i) = I - tau * v * v'
1694
1695 // where tau is a complex scalar, and v is a complex vector with
1696 // v(1:i) = 0, v(i+1) = 1 and v(ihi+1:n) = 0; v(i+2:ihi) is stored on
1697 // exit in A(i+2:ihi,i), and tau in TAU(i).
1698
1699 // The contents of A are illustrated by the following example, with
1700 // n = 7, ilo = 2 and ihi = 6:
1701
1702 // on entry, on exit,
1703
1704 // ( a a a a a a a ) ( a a h h h h a )
1705 // ( a a a a a a ) ( a h h h h a )
1706 // ( a a a a a a ) ( h h h h h h )
1707 // ( a a a a a a ) ( v2 h h h h h )
1708 // ( a a a a a a ) ( v2 v3 h h h h )
1709 // ( a a a a a a ) ( v2 v3 v4 h h h )
1710 // ( a ) ( a )
1711
1712 // where a denotes an element of the original matrix A, h denotes a
1713 // modified element of the upper Hessenberg matrix H, and vi denotes an
1714 // element of the vector defining H(i).
1715
1716 for(size_t i = ilo; i < ihi; ++i){
1717 // Compute elementary reflector H(i) to annihilate A(i+2:ihi,i)
1718 T alpha = a[i+1+i*lda];
1719 size_t row = i+2; if((int)n-1 < (int)row){ row = n-1; }
1720 RNP::TLASupport::GenerateElementaryReflector(ihi-i, &alpha, &a[row+i*lda], 1, &tau[i]);
1721 a[i+1+i*lda] = 1;
1722 // Apply H(i) to A(1:ihi,i+1:ihi) from the right */
1723 RNP::TLASupport::ApplyElementaryReflector<'R'>(ihi+1, ihi-i, &a[i+1+i*lda], 1, tau[i], &a[0+(i+1)*lda], lda, work);
1724 // Apply H(i)' to A(i+1:ihi,i+1:n) from the left */
1725 RNP::TLASupport::ApplyElementaryReflector<'L'>(ihi-i, n-i-1, &a[i+1+i*lda], 1, std::conj(tau[i]), &a[i+1+(i+1)*lda], lda, work);
1726 a[i+1+i*lda] = alpha;
1727 }
1728}

1730template <class T> // zlahr2, dlahr2, clahr2, slahr2
1731void HessenbergReductionBlocked(size_t n, size_t k, size_t nb, T *a, size_t lda, T *tau, T *t, size_t ldt, T *y, size_t ldy){
1732 if(n <= 1){ return; }
1733 T ei = 0;
1734 for(size_t i = 0; i < nb; ++i){
1735 if(i > 0){
1736 // Update A(K+1:N,I)
1737 // Update I-th column of A - Y * V'
1738 RNP::TBLAS::Conjugate(i, &a[k+i+0*lda], lda);
1739 RNP::TBLAS::MultMV<'N'>(n-k, i, -1., &y[k+0*ldy], ldy, &a[k+i+0*lda], lda, 1., &a[k+(i-1)*lda], 1);
1740 RNP::TBLAS::Conjugate(i, &a[k+i+0*lda], lda);
1741 // Apply I - V * T' * V' to this column (call it b) from the
1742 // left, using the last column of T as workspace
1743 //
1744 // Let V = ( V1 ) and b = ( b1 ) (first I-1 rows)
1745 // ( V2 ) ( b2 )
1746 //
1747 // where V1 is unit lower triangular
1748 //
1749 // w := V1' * b1
1750 RNP::TBLAS::Copy(i, &a[k+i*lda], 1, &t[0+(nb-1)*ldt], 1);
1751 RNP::TBLAS::MultTrV<'L','C','U'>(i, &a[k+0*lda], lda, &t[0+(nb-1)], 1);
1752 // w := w + V2'*b2
1753 RNP::TBLAS::MultMV<'C'>(n-k-i, i, 1., &a[k+i+0*lda], lda, &a[k+i+i*lda], 1, 1., &t[0+(nb-1)*ldt], 1);
1754 // w := T'*w
1755 RNP::TBLAS::MultTrV<'U','C','N'>(i, t,ldt, &t[0+(nb-1)*ldt], 1);
1756 // b2 := b2 - V2*w
1757 RNP::TBLAS::MultMV<'N'>(n-k-i, i, -1., &a[k+i+0*lda], lda, &t[0+(nb-1)*ldt], 1, 1., &a[k+i+i*lda], 1);
1758 // b1 := b1 - V1*w
1759 RNP::TBLAS::MultTrV<'L','N','U'>(i, &a[k+0*lda], lda, &t[0+(nb-1)*ldt], 1);
1760 RNP::TBLAS::Axpy(i, -1., &t[0+(nb-1)*ldt], 1, &a[k+i*lda], 1);
1761 a[k+i-1+(i-1)*lda] = ei;
1762 }
1763 // Generate the elementary reflector H(I) to annihilate
1764 // A(K+I+1:N,I)
1765 size_t arow = (k+i+2 < n ? k+i+1 : n-1);
1766 GenerateElementaryReflector(n-k-i, &a[k+i+i*lda], &a[arow+i*lda], 1, &tau[i]);
1767 ei = a[k+i+i*lda];
1768 a[k+i+i*lda] = 1.;
1769 // Compute Y(K+1:N,I)
1770 RNP::TBLAS::MultMV<'N'>(n-k, n-k-i, 1., &a[k+(i+1)*lda], lda, &a[k+i+i*lda], 1, 0., &y[k+i*ldy], 1);
1771 RNP::TBLAS::MultMV<'C'>(n-k-i, i, 1., &a[k+i+0*lda], lda, &a[k+i+i*lda], 1, 0., &t[0+i*ldt], 1);
1772 RNP::TBLAS::MultMV<'N'>(n-k, i, -1., &y[k+0*ldy], ldy, &t[0+i*ldt], 1, 1., &y[k+i*ldy], 1);
1773 RNP::TBLAS::Scale(n-k, tau[i], &y[k+i*ldy], 1);
1774 // Compute T(1:I,I)
1775 RNP::TBLAS::Scale(i, -tau[i], &t[0+i*ldt], 1);
1776 RNP::TBLAS::MultTrV<'U','N','N'>(i, t, ldt, &t[0+i*ldt], 1);
1777 t[i+i*ldt] = tau[i];
1778 }
1779 a[k+nb-1+(nb-1)*lda] = ei;
1780 // Compute Y(1:K,1:NB)
1781 RNP::TBLAS::CopyMatrix<'A'>(k, nb, &a[0+1*lda],lda, y,ldy);
1782 RNP::TBLAS::MultTrM<'R','L','N','U'>(k, nb, 1., &a[k+0*lda], lda, y, ldy);
1783 if(n > k+nb){
1784 RNP::TBLAS::MultMM<'N','N'>(k, nb, n-k-nb, 1., &a[0+(2+nb)*lda], lda, &a[k+nb+0*lda], lda, 1., y, ldy);
1785 }
1786 RNP::TBLAS::MultTrM<'R','U','N','N'>(k, nb, 1., t, ldt, y, ldy);
1787}

1789template <class T> // zgehrd, dgehrd, cgehrd, sgehrd
1790void HessenbergReduction(size_t n, size_t ilo, size_t ihi, T *a, size_t lda, T *tau, T *work, size_t lwork){
1791 //
1792 // -- LAPACK routine (version 3.2.1) --
1793 // -- LAPACK is a software package provided by Univ. of Tennessee, --
1794 // -- Univ. of California Berkeley, Univ. of Colorado Denver and NAG Ltd..--
1795 // -- April 2009 --
1796 //
1797 //
1798 // Purpose
1799 // =======
1800 //
1801 // ZGEHRD reduces a complex general matrix A to upper Hessenberg form H by
1802 // an unitary similarity transformation: Q' * A * Q = H .
1803 //
1804 // Arguments
1805 // =========
1806 //
1807 // N (input) INTEGER
1808 // The order of the matrix A. N >= 0.
1809 //
1810 // ILO (input) INTEGER
1811 // IHI (input) INTEGER
1812 // It is assumed that A is already upper triangular in rows
1813 // and columns 1:ILO-1 and IHI+1:N. ILO and IHI are normally
1814 // set by a previous call to ZGEBAL; otherwise they should be
1815 // set to 1 and N respectively. See Further Details.
1816 // 1 <= ILO <= IHI <= N, if N > 0; ILO=1 and IHI=0, if N=0.
1817 //
1818 // A (input/output) COMPLEX*16 array, dimension (LDA,N)
1819 // On entry, the N-by-N general matrix to be reduced.
1820 // On exit, the upper triangle and the first subdiagonal of A
1821 // are overwritten with the upper Hessenberg matrix H, and the
1822 // elements below the first subdiagonal, with the array TAU,
1823 // represent the unitary matrix Q as a product of elementary
1824 // reflectors. See Further Details.
1825 //
1826 // LDA (input) INTEGER
1827 // The leading dimension of the array A. LDA >= max(1,N).
1828 //
1829 // TAU (output) COMPLEX*16 array, dimension (N-1)
1830 // The scalar factors of the elementary reflectors (see Further
1831 // Details). Elements 1:ILO-1 and IHI:N-1 of TAU are set to
1832 // zero.
1833 //
1834 // WORK (workspace/output) COMPLEX*16 array, dimension (LWORK)
1835 // On exit, if INFO = 0, WORK(1) returns the optimal LWORK.
1836 //
1837 // LWORK (input) INTEGER
1838 // The length of the array WORK. LWORK >= max(1,N).
1839 // For optimum performance LWORK >= N*NB, where NB is the
1840 // optimal blocksize.
1841 //
1842 // If LWORK = -1, then a workspace query is assumed; the routine
1843 // only calculates the optimal size of the WORK array, returns
1844 // this value as the first entry of the WORK array, and no error
1845 // message related to LWORK is issued by XERBLA.
1846 //
1847 // INFO (output) INTEGER
1848 // = 0: successful exit
1849 // < 0: if INFO = -i, the i-th argument had an illegal value.
1850 //
1851 // Further Details
1852 // ===============
1853 //
1854 // The matrix Q is represented as a product of (ihi-ilo) elementary
1855 // reflectors
1856 //
1857 // Q = H(ilo) H(ilo+1) . . . H(ihi-1).
1858 //
1859 // Each H(i) has the form
1860 //
1861 // H(i) = I - tau * v * v'
1862 //
1863 // where tau is a complex scalar, and v is a complex vector with
1864 // v(1:i) = 0, v(i+1) = 1 and v(ihi+1:n) = 0; v(i+2:ihi) is stored on
1865 // exit in A(i+2:ihi,i), and tau in TAU(i).
1866 //
1867 // The contents of A are illustrated by the following example, with
1868 // n = 7, ilo = 2 and ihi = 6:
1869 //
1870 // on entry, on exit,
1871 //
1872 // ( a a a a a a a ) ( a a h h h h a )
1873 // ( a a a a a a ) ( a h h h h a )
1874 // ( a a a a a a ) ( h h h h h h )
1875 // ( a a a a a a ) ( v2 h h h h h )
1876 // ( a a a a a a ) ( v2 v3 h h h h )
1877 // ( a a a a a a ) ( v2 v3 v4 h h h )
1878 // ( a ) ( a )
1879 //
1880 // where a denotes an element of the original matrix A, h denotes a
1881 // modified element of the upper Hessenberg matrix H, and vi denotes an
1882 // element of the vector defining H(i).
1883 //
1884 // This file is a slight modification of LAPACK-3.0's DGEHRD
1885 // subroutine incorporating improvements proposed by Quintana-Orti and
1886 // Van de Geijn (2006). (See DLAHR2.)
1887 //
1888
1889 static const size_t nbopt = 32; // block size
1890 static const size_t nbmin = 2; // minimum block size
1891 static const size_t nx_parm = 128; // crossover point from blocked to unblocked
1892
1893 T t[nbopt*nbopt];
1894 const size_t ldt = nbopt;
1895
1896 size_t nb = nbopt;
1897
1898 work[0] = n*nb;
1899 if((size_t)-1 == lwork){ return; }
1900
1901 for(size_t i = 0; i < ilo; ++i){
1902 tau[i] = 0.;
1903 }
1904 for(size_t i = ihi; i+1 < n; ++i){
1905 tau[i] = 0.;
1906 }
1907 const size_t nh = ihi - ilo + 1;
1908 if(nh <= 1){ work[0] = 1; return; }
1909
1910 size_t iws = 0;
1911
1912 const size_t nx = nb > nx_parm ? nb : nx_parm;
1913 if(1 < nb && nb < nh){
1914 // determine cross over point
1915 if(nx < nh){
1916 iws = n*nb;
1917 if(lwork < iws){ // not enough workspace
1918 if(lwork >= n*nbmin){
1919 nb = lwork/n;
1920 }else{
1921 nb = 1;
1922 }
1923 }
1924 }
1925 }
1926
1927 size_t i = ilo;
1928
1929 if(nb < nbmin || nb >= nh){
1930 }else{ // use blocked code
1931 for(i = ilo; i+nx+1 <= ihi; i += nb){
1932 size_t ib = (nb < (ihi-i) ? nb : (ihi-i));
1933 // Reduce columns i:i+ib-1 to Hessenberg form, returning the
1934 // matrices V and T of the block reflector H = I - V*T*V'
1935 // which performs the reduction, and also the matrix Y = A*V*T
1936 HessenbergReductionBlocked(ihi, i, ib, &a[0+i*lda],lda, &tau[i], t, ldt, work, n);
1937 // Apply the block reflector H to A(1:ihi,i+ib:ihi) from the
1938 // right, computing A := A - Y * V'. V(i+ib,ib-1) must be set to 1
1939 T ei = a[i+ib+(i+ib-1)*lda];
1940 a[i+ib+(i+ib-1)*lda] = 1.;
1941 RNP::TBLAS::MultMM<'N','C'>(ihi+1, ihi-i-ib+1+1, ib+1, -1., work,n, &a[i+ib+i*lda],lda, 1., &a[0+(i+ib)*lda],lda);
1942 a[i+ib+(i+ib-1)*lda] = ei;
1943 // Apply the block reflector H to A(1:i,i+1:i+ib-1) from the right
1944 RNP::TBLAS::MultTrM<'R','L','C','U'>(i+1, ib-1+1, 1., &a[i+1+i*lda],lda, work,n);
1945 for(size_t j = 0; j < ib-1; ++j){
1946 RNP::TBLAS::Axpy(i+1, -1., &work[n*j],1, &a[0+(i+j)*lda],1);
1947 }
1948 // Apply the block reflector H to A(i+1:ihi,i+ib:n) from the left
1949 ApplyElementaryReflectorBlocked<'L','C','F','C'>(ihi-i+1, n-i-ib+1, ib, &a[i+1+i*lda], lda, t, ldt, &a[i+1+(i+ib)*lda], lda, work, n);
1950 }
1951 }
1952
1953 HessenbergReductionUnblocked(n, i, ihi, a, lda, tau, work);
1954 work[0] = iws;
1955}

1957template <char compq='N', char compz='N'>
1958struct GeneralizedHessenbergReduction{ // zgghrd, dgghrd, cgghrd, sgghrd
1959 template <class T>
1960 GeneralizedHessenbergReduction(size_t n, size_t ilo, size_t ihi,
1961 T *a, size_t lda, T *b, size_t ldb, T *q, size_t ldq, T *z, size_t ldz)
1962 {
1963 using namespace std;
1964
1965 // Purpose
1966 // =======
1967
1968 // ZGGHRD reduces a pair of complex matrices (A,B) to generalized upper
1969 // Hessenberg form using unitary transformations, where A is a
1970 // general matrix and B is upper triangular. The form of the
1971 // generalized eigenvalue problem is
1972 // A*x = lambda*B*x,
1973 // and B is typically made upper triangular by computing its QR
1974 // factorization and moving the unitary matrix Q to the left side
1975 // of the equation.
1976
1977 // This subroutine simultaneously reduces A to a Hessenberg matrix H:
1978 // Q**H*A*Z = H
1979 // and transforms B to another upper triangular matrix T:
1980 // Q**H*B*Z = T
1981 // in order to reduce the problem to its standard form
1982 // H*y = lambda*T*y
1983 // where y = Z**H*x.
1984
1985 // The unitary matrices Q and Z are determined as products of Givens
1986 // rotations. They may either be formed explicitly, or they may be
1987 // postmultiplied into input matrices Q1 and Z1, so that
1988 // Q1 * A * Z1**H = (Q1*Q) * H * (Z1*Z)**H
1989 // Q1 * B * Z1**H = (Q1*Q) * T * (Z1*Z)**H
1990 // If Q1 is the unitary matrix from the QR factorization of B in the
1991 // original equation A*x = lambda*B*x, then ZGGHRD reduces the original
1992 // problem to generalized Hessenberg form.
1993
1994 // Arguments
1995 // =========
1996
1997 // COMPQ (input) CHARACTER*1
1998 // = 'N': do not compute Q;
1999 // = 'I': Q is initialized to the unit matrix, and the
2000 // unitary matrix Q is returned;
2001 // = 'V': Q must contain a unitary matrix Q1 on entry,
2002 // and the product Q1*Q is returned.
2003
2004 // COMPZ (input) CHARACTER*1
2005 // = 'N': do not compute Q;
2006 // = 'I': Q is initialized to the unit matrix, and the
2007 // unitary matrix Q is returned;
2008 // = 'V': Q must contain a unitary matrix Q1 on entry,
2009 // and the product Q1*Q is returned.
2010
2011 // N (input) INTEGER
2012 // The order of the matrices A and B. N >= 0.
2013
2014 // ILO (input) INTEGER
2015 // IHI (input) INTEGER
2016 // ILO and IHI mark the rows and columns of A which are to be
2017 // reduced. It is assumed that A is already upper triangular
2018 // in rows and columns 1:ILO-1 and IHI+1:N. ILO and IHI are
2019 // normally set by a previous call to ZGGBAL; otherwise they
2020 // should be set to 1 and N respectively.
2021 // 1 <= ILO <= IHI <= N, if N > 0; ILO=1 and IHI=0, if N=0.
2022
2023 // A (input/output) COMPLEX*16 array, dimension (LDA, N)
2024 // On entry, the N-by-N general matrix to be reduced.
2025 // On exit, the upper triangle and the first subdiagonal of A
2026 // are overwritten with the upper Hessenberg matrix H, and the
2027 // rest is set to zero.
2028
2029 // LDA (input) INTEGER
2030 // The leading dimension of the array A. LDA >= max(1,N).
2031
2032 // B (input/output) COMPLEX*16 array, dimension (LDB, N)
2033 // On entry, the N-by-N upper triangular matrix B.
2034 // On exit, the upper triangular matrix T = Q**H B Z. The
2035 // elements below the diagonal are set to zero.
2036
2037 // LDB (input) INTEGER
2038 // The leading dimension of the array B. LDB >= max(1,N).
2039
2040 // Q (input/output) COMPLEX*16 array, dimension (LDQ, N)
2041 // On entry, if COMPQ = 'V', the unitary matrix Q1, typically
2042 // from the QR factorization of B.
2043 // On exit, if COMPQ='I', the unitary matrix Q, and if
2044 // COMPQ = 'V', the product Q1*Q.
2045 // Not referenced if COMPQ='N'.
2046
2047 // LDQ (input) INTEGER
2048 // The leading dimension of the array Q.
2049 // LDQ >= N if COMPQ='V' or 'I'; LDQ >= 1 otherwise.
2050
2051 // Z (input/output) COMPLEX*16 array, dimension (LDZ, N)
2052 // On entry, if COMPZ = 'V', the unitary matrix Z1.
2053 // On exit, if COMPZ='I', the unitary matrix Z, and if
2054 // COMPZ = 'V', the product Z1*Z.
2055 // Not referenced if COMPZ='N'.
2056
2057 // LDZ (input) INTEGER
2058 // The leading dimension of the array Z.
2059 // LDZ >= N if COMPZ='V' or 'I'; LDZ >= 1 otherwise.
2060
2061 // Further Details
2062 // ===============
2063
2064 // This routine reduces A to Hessenberg and B to triangular form by
2065 // an unblocked reduction, as described in _Matrix_Computations_,
2066 // by Golub and van Loan (Johns Hopkins Press).
2067
2068 const bool ilq = ((compq == 'V') || (compq == 'I'));
2069 const bool ilz = ((compz == 'V') || (compz == 'I'));
2070
2071 // Initialize Q and Z if desired.
2072 if(compq == 'I'){
2073 RNP::TBLAS::SetMatrix<'F'>(n, n, 0, 1, q, ldq);
2074 }
2075 if(compz == 'I'){
2076 RNP::TBLAS::SetMatrix<'F'>(n, n, 0, 1, z, ldz);
2077 }
2078
2079 if(n <= 1){ return; }
2080
2081 // Zero out lower triangle of B
2082 for(size_t jcol = 0; jcol < n-1; ++jcol){
2083 for(size_t jrow = jcol+1; jrow < n; ++jrow){
2084 b[jrow+jcol*ldb] = 0;
2085 }
2086 }
2087
2088 // Reduce A and B
2089 for(size_t jcol = ilo; (int)jcol <= (int)ihi - 2; ++jcol){
2090 for(size_t jrow = ihi; jrow >= jcol + 2; --jrow){
2091 typename RNP::TBLAS::_RealOrComplexChooser<T>::real_type c;
2092 T s, ctemp;
2093
2094 // Step 1: rotate rows JROW-1, JROW to kill A(JROW,JCOL)
2095 ctemp = a[jrow-1+jcol*lda];
2096 RNP::TLASupport::GeneratePlaneRotation(ctemp, a[jrow+jcol*lda], &c, &s, &a[jrow-1+jcol*lda]);
2097 a[jrow+jcol*lda] = 0;
2098 RNP::TLASupport::ApplyPlaneRotation(n-jcol-1, &a[jrow-1+(jcol+1) * lda], lda, &a[jrow+(jcol+1)*lda], lda, c, s);
2099 RNP::TLASupport::ApplyPlaneRotation(n+1-jrow, &b[jrow-1+(jrow-1)*ldb], ldb, &b[jrow+(jrow-1)*ldb], ldb, c, s);
2100 if(ilq){
2101 RNP::TLASupport::ApplyPlaneRotation(n, &q[0+(jrow-1)*ldq], 1, &q[0+jrow*ldq], 1, c, std::conj(s));
2102 }
2103
