100typedef struct Options_{
101 int use_discretized_epsilon;
102
103 // Set use_subpixel_smoothing to nonzero if subpixel smoothing should
104 // be done on the Fourier transform of the dielectric functions.
105 // This will cause the dielectric function to be discretized instead
106 // of using the analytic Fourier transforms of specified shapes.
107 int use_subpixel_smoothing;
108
109 // Set use_Lanczos_smoothing to nonzero if Lanczos smoothing should
110 // be done on the Fourier transform of the dielectric functions.
111 int use_Lanczos_smoothing;
112
113 // Set use_polarization_basis to nonzero if a decomposition into the
114 // structure conforming polarization basis should be performed. By
115 // default, the basis is not everywhere-unit-normalized and purely
116 // real unless one of the following two options are used.
117 int use_polarization_basis;
118
119 // Set use_jones_vector_basis to nonzero if a complex Jones
120 // vector polarization basis should be used. This setting has no
121 // effect if use_polarization_basis is zero.
122 int use_jones_vector_basis;
123
124 // Set use_normal_vector_basis to nonzero if a normalized normal
125 // vector polarization basis should be used. This setting has no
126 // effect if use_polarization_basis is zero.
127 int use_normal_vector_basis;
128
129 // Set use_normal_vector_field to nonzero if a normal vector field
130 // should be generated. The field is everywhere normal to material
131 // interfaces and forced to have unit magnitude everywhere.
132 int use_normal_vector_field;
133
134 // Set resolution to be the multiple of the reciprocal lattice order
135 // extent by which the polarization basis vector field should be
136 // discretized. For example, if the reciprocal lattice orders go up
137 // to G = {2,1}, then the vector field will be generated on a grid
138 // of dimension 2*f x 1*f, where f is this value.
139 // It is preferable to set this value to have small integer factors
140 // in order to make the FFT more efficient. Powers of 2 are best.
141 // The default value is 64, and must be at least 2 to satisfy the
142 // Nyquist sampling criterion.
143 int resolution;
144
145 // Set vector_field_dump_filename_prefix to non-NULL if the
146 // automatically generated vector field in the polarization
147 // decomposition should be output to files whose prefix is the
148 // specified string (suffix is layer name).
149 char *vector_field_dump_filename_prefix;
150
151 // Specifies how G vectors should be chosen (currently not used)
152 // 0 = Circular truncation
153 // 1 = Parallelogramic truncation
154 int lattice_truncation;
155
156 // Specifies how much simulation output progress to output to stdout
157 int verbosity;
158
159 // Set use_experimental_fmm to nonzero if the experimental FMM
160 // formulation should be used. (For experimental/debug purposes)
161 int use_experimental_fmm;
162
163 // Set use_less_memory to nonzero if intermediate matrices should not
164 // be stored in memory when possible.
165 int use_less_memory;
166
167 // Use Weismann's improved formulation for the fields
168 int use_weismann_formulation;
169
170 double lanczos_smoothing_width;
171 int lanczos_smoothing_power;
172} Options;

212typedef struct Simulation_{
213 double Lr[4]; // real space lattice:
214 // {Lr[0],Lr[1]} is the first basis vector's x and y coords.
215 // {Lr[2],Lr[3]} is the second basis vector's x and y coords.
216 double Lk[4]; // reciprocal lattice:
217 // 2pi*{Lk[0],Lk[1]} is the first basis vector's x and y coords.
218 // 2pi*{Lk[2],Lk[3]} is the second basis vector's x and y coords.
219 // Computed by taking the inverse of Lr as a 2x2 column-major matrix.
220 int n_G; // Number of G-vectors (Fourier planewave orders).
221 Material *material; // linked list of materials
222 Layer *layer; // linked list of layers
223
224 // Excitation
225 double omega[2]; // real and imaginary parts of omega
226 double k[2]; // xy components of k vector, as fraction of k0 = omega
227 Excitation exc;
228
229 Solution *solution; // The solution object is not allocated until needed.
230 Options options;
231
232 struct FieldCache *field_cache; // Internal cache of vector field FT when using polarization bases
233} Simulation;
