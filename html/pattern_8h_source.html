23/* A pattern is defined as a set of 1D geometrical shapes (e.g. circles)
24 * in the 2D plane. All shapes must be closed, simply connected contours.
25 *
26 * Available shapes:
27 * circle
28 * ellipse
29 * rectangle
30 * polygon
31 *
32 * A valid pattern must not have any shapes that intersect with each other.
33 * Due to this requirement, each shape is either contained entirely within
34 * another shape, or has no containing shape. The set of shapes then forms
35 * a forest (a set of trees), where the parent-child relationship within a
36 * tree is the immediate container-containee relationship.
37 *
38 * Currently, the no-intersection criterion is not checked due its
39 * complexity, so the input shapes must be sanitized elsewhere.
40 */
41typedef enum shape_type_{
42 CIRCLE,
43 ELLIPSE,
44 RECTANGLE,
45 POLYGON
46} shape_type;
47
48typedef struct shape_{
49 /* `type' determines which field in the union is used. */
50 shape_type type;
51
52 /* Location of the center of the shape. For circles, ellipses, and
53 * rectangles, the center is the geometric center (center of mass).
54 */
55 double center[2]; /* x and y coordinates */
56
57 /* The angle by which the shape is rotated, CCW, in radians.
58 * Obviously, the rotation is applied first, then the translation
59 * given in `center' is applied.
60 */
61 double angle;
62
63 /* The proper choice of field to use is determined by `type' above.
64 */
65 union{
66 struct{
67 double radius;
68 } circle;
69 struct{
70 double halfwidth[2]; /* x and y semimajor axes */
71 } ellipse;
72 struct{
73 double halfwidth[2]; /* x and y halfwidths */
74 } rectangle;
75 struct{
76 int n_vertices;
77 /* Separately allocated, length 2*n_vertices.
78 * A list of xy-pairs in CCW orientation.
79 */
80 double *vertex;
81 } polygon;
82 } vtab;
83 int tag; /* available for user defined purposes */
84} shape;

96/* Computes the containment relationship of all the shapes.
97 * The shapes must not intersect one another, so that any shape is either
98 * not contained within any other shape, or has a unique immediately
99 * containing shape. The containing shape of shapes[i] is shapes[parent[i]]
100 * unless parent[i] == -1, in which case shapes[i] is not contained within
101 * any other shape.
102 *
103 * Arguments:
104 * nshapes IN Number of shapes; length of `shapes'.
105 * shapes IN/OUT Array of `shape' structures. On exit, the array
106 * is sorted in order of decreasing area of shapes.
107 * parent OUT Length `nshapes'. Gives the immediate containing
108 * shape of each shape, or -1 if none.
109 * Return values:
110 * 0: If no shapes intersect each other and containment tree
111 * returned is valid.
112 * 0 < n <= nshapes : If shapes[n-1] is invalid.
113 * nshapes < n <= 2*nshapes: If shapes[n-1] intersects with another shape.
114 * -n: If n-th argument is invalid.
115 */
116int pattern_get_containment_tree(
117 int nshapes,
118 shape *shapes,
119 int *parent
120);

124int Pattern_GetContainmentTree(

174/* Returns the Fourier transform of the pattern, given values for the interior
175 * of each shape and the background.
176 * The Fourier transform convention is
177 * F(kx,ky) = Integral f(x,y) exp(-i*(kx*x + ky*y)) dx dy
178 *
179 * Arguments:
180 * nshapes IN Number of shapes; length of `shapes'.
181 * shapes IN Array of `shape' structures in order of decreasing
182 * area. (as output from pattern_get_containment_tree).
183 * parent IN Immediate containing shape relationship for each
184 * shape. (as output from pattern_get_containment_tree).
185 * value IN Length `2*(nshapes+1)'. Real and imaginary parts of
186 * of the interior value of each shape. The background
187 * value is `value[0]+i*value[1]', and the value of
188 * shape[i] is `value[2*(i+1)+0]+i*value[2*(i+1)+1]'.
189 * f IN Point in reciprocal space to obtain the F.T., but
190 * coordinates should be divided by 2*pi (e.g. On a unit
191 * square real space lattice, the point (1,0) should be
192 * used instead of (2*pi,0).
193 * unit_cell_size IN Area of the unit cell in real space.
194 * FT OUT Real and imaginary part of the F.T. at `2*pi*f'.
195 * Return values:
196 * 0: If successful.
197 * -n: If n-th argument is invalid.
198 */

219/* Returns an area weighting of each shape within one cell of a uniform
220 * discretization of the origin-centered unit square.
221 * The area-fraction of each shape within the rectangle
222 * [x0,x0+dx] x [y0,y0+dy] is returned in `value'. The discretization has
223 * `nx' cells along the x-direction, and `ny' cells along the y-direction.
224 * The total region is the unit square [-0.5,0.5] x [-0.5,0.5], so
225 * x0 = -0.5, 0.5 + dx, 0.5 + 2*dx, ..., 0.5 - dx
226 * where dx = 1/nx. Similarly for y0.
227 *
228 * Arguments:
229 * nshapes IN Number of shapes; length of `shapes'.
230 * shapes IN Array of `shape' structures in order of decreasing
231 * area. (as output from pattern_get_containment_tree).
232 * parent IN Immediate containing shape relationship for each
233 * shape. (as output from pattern_get_containment_tree).
234 * L IN Lattice basis vectors, (L[0],L[1]) are x- and y-
235 * coordinates of the first (u) basis vector, and
236 * (L[2],L[3]) are the coordinates of the second (v)
237 * basis vector.
238 * nu, nv IN Number of cells in u and v directions. nu > 0, nv > 0.
239 * iu, iv IN u and v index of the cell at which the area fraction
240 * is to be computed. 0 <= iu < nu, 0 <= iv < nv.
241 * value OUT Length `nshapes+1'. value[0] is the area fraction of
242 * the background (no shape). The area fraction of
243 * shape[i] is stored in value[i+1].
244 * Return values:
245 * 0: If successful.
246 * -n: If n-th argument is invalid.
247 */

266/* Returns a vector field which is tangential or normal to the shapes.
267 * The field is generated as the minimizer of the Dirichlet 1-form energy
268 * on the specified uniform rectangular grid subject to the constraints
269 * that the resulting vector field be equal to the tangent or normal vectors
270 * of the parameterizations of the shape contours, with periodic boundary
271 * conditions over the origin-centered unit square.
272 *
273 * Arguments:
274 * nshapes IN Number of shapes; length of `shapes'.
275 * shapes IN Array of `shape' structures in order of decreasing
276 * area. (as output from pattern_get_containment_tree).
277 * parent IN Immediate containing shape relationship for each
278 * shape. (as output from pattern_get_containment_tree).
279 * type IN 0 for a field tangential to shapes, 1 for normal using
280 * shape centers as sources.
281 * L IN Lattice basis vectors, (L[0],L[1]) are x- and y-
282 * coordinates of the first (u) basis vector, and
283 * (L[2],L[3]) are the coordinates of the second (v)
284 * basis vector.
285 * nu, nv IN Number of cells in u and v directions. nu > 0, nv > 0.
286 * field OUT Length 2*nu*nv, array of xy-vectors. The field vector
287 * located at uv-index (i,j) is
288 * { field[2*(i+j*nu)+0], field[2*(i+j*nu)+1] }
289 * Return values:
290 * 0: If successful.
291 * 1: If the discretization is not fine enough to properly apply
292 * constraints, although a best-attempt field was generated.
293 * -n: If the n-th argument is invalid.
294 *
295 * Notes:
296 * The details of implementation depend on the lattice type.
297 * In the case of an orthogonal lattice, we use a rectangular quad mesh.
298 * We use an interpolation scheme on the dual mesh to compute the
299 * constraints, which is piecewise constant over each edge. The Laplacian
300 * is then just a generalized (4,-1,-1,-1,-1) stencil.
301 * In the case of a non-orthogonal lattice, we split the parallelogramic
302 * discretization cells along the short diagonal into two triangles. We
303 * assume that the triangulation is well centered and use a circumcentric
304 * dual mesh with a diagonal Hodge star to construct the Laplacian. The
305 * constraints are enforced using the primal Whitney 1-form interpolator.
306 * In both cases, the resulting field is computed by the proper
307 * interpolation scheme at the centers of the parallelogramic discretization
308 * cells. The constraints are enforced using weighted least squares with
309 * uniform weights per edge (not per edge-shape intersection).
310 * To deal with shapes extending beyond the origin-centered fundamental
311 * parallelogram of the lattice, edge-shape intersections are computed using
312 * each shape and its 8 nearest neighbor unit cell copies.
313 */

315 int nshapes,

320 int nu, int nv,
321 double *field
