42/* a vector integrand - evaluates the function at the given point x
43 (an array of length ndim) and returns the result in fval (an array
44 of length fdim). The void* parameter is there in case you have
45 to pass any additional data through to your function (it corresponds
46 to the fdata parameter you pass to adapt_integrate). */
47typedef void (*integrand) (unsigned ndim, const double *x, void *,
48 unsigned fdim, double *fval);
49
50/* a vector integrand of a vector of npt points: x[i*ndim + j] is the
51 j-th coordinate of the i-th point, and the k-th function evaluation
52 for the i-th point is returned in fval[k*npt + i]. */
53typedef void (*integrand_v) (unsigned ndim, unsigned npt,
54 const double *x, void *,
55 unsigned fdim, double *fval);

57/* Integrate the function f from xmin[dim] to xmax[dim], with at most
58 maxEval function evaluations (0 for no limit), until the given
59 absolute or relative error is achieved. val returns the integral,
60 and err returns the estimate for the absolute error in val; both
61 of these are arrays of length fdim, the dimension of the vector
62 integrand f(x). The return value of the function is 0 on success
63 and non-zero if there was an error. */
64int adapt_integrate(unsigned fdim, integrand f, void *fdata,
65 unsigned dim, const double *xmin, const double *xmax,
66 unsigned maxEval, double reqAbsError, double reqRelError,
67 double *val, double *err);
