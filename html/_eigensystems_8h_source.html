24// Eigensystem computes for an N-by-N complex nonsymmetric matrix A, the
25// eigenvalues and, optionally, the left and/or right eigenvectors.
26//
27// The right eigenvector v(j) of A satisfies
28// A * v(j) = lambda(j) * v(j)
29// where lambda(j) is its eigenvalue.
30// The left eigenvector u(j) of A satisfies
31// u(j)**H * A = lambda(j) * u(j)**H
32// where u(j)**H denotes the conjugate transpose of u(j).
33//
34// The computed eigenvectors are normalized to have Euclidean norm
35// equal to 1 and largest component real.
36//
37// Arguments
38// =========
39//
40// n The order of the matrix A. n >= 0.
41//
42// a (input/output) dimension (lda,n)
43// On entry, the N-by-N matrix A.
44// On exit, A has been overwritten.
45//
46// lda The leading dimension of the array A. lda >= n.
47//
48// eval (output) dimension (n)
49// eval contains the computed eigenvalues.
50//
51// vl (output) dimension (ldvl,n)
52// If not NULL, the left eigenvectors u(j) are stored one
53// after another in the columns of vl, in the same order
54// as their eigenvalues.
55// u(j) = vl(:,j), the j-th column of vl.
56//
57// ldvl The leading dimension of the array vl. If vl != NULL, ldvl >= N.
58//
59// vr (output) dimension (ldvr,n)
60// If not NULL, the right eigenvectors v(j) are stored one
61// after another in the columns of vr, in the same order
62// as their eigenvalues.
63// v(j) = vr(:,j), the j-th column of vr.
64//
65// ldvr The leading dimension of the array vr. vr != NULL, ldvr >= n.
66//
67// work dimension 2*N
68// rwork dimension 2*N
69//
70// return = 0: successful exit
71// < 0: if INFO = -i, the i-th argument had an illegal value.
72// > 0: if INFO = i, the QR algorithm failed to compute all the
73// eigenvalues, and no eigenvectors have been computed;
74// elements and i+1:N of W contain eigenvalues which have
75// converged.
76int Eigensystem(size_t n,
77 std::complex<double> *a, size_t lda,
78 std::complex<double> *eval,
79 std::complex<double> *vl, size_t ldvl, std::complex<double> *vr, size_t ldvr,
80 std::complex<double> *work = NULL, double *rwork = NULL, size_t lwork = 0);
